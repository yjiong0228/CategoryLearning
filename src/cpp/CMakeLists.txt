cmake_minimum_required(VERSION 3.24)
project(CategoryLearning1031Cpp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(FETCHCONTENT_QUIET OFF)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/v3.11.3 URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406)
FetchContent_Declare(eigen URL https://codeload.github.com/eigen-mirror/eigen/tar.gz/3.4.0 URL_HASH SHA256=8586084f71f9bde545ee7fa6d00288b264a2b7ac3607b974e54d13e7162c1c72)
FetchContent_Declare(yaml URL https://codeload.github.com/jbeder/yaml-cpp/tar.gz/0.8.0 URL_HASH SHA256=fbe74bbdcee21d656715688706da3c8becfd946d92cd44705cc6098bb23b3a16)
FetchContent_Declare(zlib URL https://codeload.github.com/madler/zlib/tar.gz/v1.3.1 URL_HASH SHA256=17e88863f3600672ab49182f217281b6fc4d3c762bde361935e436a95214d05c)
FetchContent_Declare(cnpy URL https://codeload.github.com/rogersce/cnpy/tar.gz/4e8810b1a8637695171ed346ce68f6984e585ef4 URL_HASH SHA256=5120abc54a564efa92c642cc0199cc4fd3f345901157de9fbbdcedbb34d28d8a)
FetchContent_Declare(csv URL https://codeload.github.com/vincentlaucsb/csv-parser/tar.gz/2.3.0 URL_HASH SHA256=27b8ac51aa58b9a4debd8ccfb44738c8583a2e874da42f56bbdf3764b75f3af5)
FetchContent_Declare(sha URL https://codeload.github.com/okdshin/PicoSHA2/tar.gz/161cb3fc4170fa7a3eca9e582cebd27cc4d1fe29 URL_HASH SHA256=6cf473a00c98298d3ddee0aed853e3c799791f49dbc01c996ea46cb248e85802)
FetchContent_Declare(doctest URL https://codeload.github.com/doctest/doctest/tar.gz/v2.4.11 URL_HASH SHA256=632ed2c05a7f53fa961381497bf8069093f0d6628c5f26286161fbd32a560186)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json yaml zlib doctest)
foreach(dep eigen cnpy csv sha)
  FetchContent_GetProperties(${dep})
  if(NOT ${dep}_POPULATED)
    FetchContent_Populate(${dep})
  endif()
endforeach()
find_package(OpenMP REQUIRED)
add_library(gm_npz STATIC ${cnpy_SOURCE_DIR}/cnpy.cpp)
target_compile_definitions(gm_npz PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(gm_npz PUBLIC ${cnpy_SOURCE_DIR} ${zlib_SOURCE_DIR} ${zlib_BINARY_DIR})
target_link_libraries(gm_npz PUBLIC zlibstatic)
file(GLOB GM_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(FILTER GM_SOURCES EXCLUDE REGEX "/cli\\.cpp$")
add_library(gm_core STATIC ${GM_SOURCES})
target_include_directories(gm_core PUBLIC include)
target_include_directories(gm_core SYSTEM PUBLIC ${eigen_SOURCE_DIR})
target_include_directories(gm_core SYSTEM PRIVATE ${csv_SOURCE_DIR}/single_include ${sha_SOURCE_DIR})
target_link_libraries(gm_core PUBLIC nlohmann_json::nlohmann_json yaml-cpp::yaml-cpp gm_npz OpenMP::OpenMP_CXX)
file(GLOB_RECURSE GM_IDENTITY_FILES CONFIGURE_DEPENDS src/*.cpp include/*.hpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${GM_IDENTITY_FILES})
set(GM_SOURCE_HASH "")
foreach(source IN LISTS GM_IDENTITY_FILES)
  file(SHA256 "${source}" digest)
  string(APPEND GM_SOURCE_HASH "${digest}")
endforeach()
string(SHA256 GM_SOURCE_HASH "${GM_SOURCE_HASH}")
target_compile_definitions(gm_core PRIVATE GM_SOURCE_HASH="${GM_SOURCE_HASH}" GM_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" GM_BUILD_MODE="$<CONFIG>")
if(MSVC)
  target_compile_options(gm_core PRIVATE /W4 /utf-8 /fp:precise /bigobj)
else()
  target_compile_options(gm_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/cli.cpp)
  add_executable(genmodel_cpp src/cli.cpp)
  target_link_libraries(genmodel_cpp PRIVATE gm_core)
  if(MSVC)
    target_compile_options(genmodel_cpp PRIVATE /utf-8 /bigobj)
  endif()
endif()
add_executable(gm_benchmark tools/native_benchmark.cpp)
target_link_libraries(gm_benchmark PRIVATE gm_core)
if(WIN32)
  target_link_libraries(gm_benchmark PRIVATE psapi)
endif()
enable_testing()
file(GLOB GM_TESTS CONFIGURE_DEPENDS tests/test_*.cpp)
add_executable(gm_tests tests/main.cpp ${GM_TESTS})
target_link_libraries(gm_tests PRIVATE gm_core doctest::doctest)
target_compile_definitions(gm_tests PRIVATE GM_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(gm_tests PRIVATE /utf-8 /bigobj)
endif()
add_test(NAME core COMMAND gm_tests)
set_tests_properties(core PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} ENVIRONMENT "OMP_NUM_THREADS=1")
